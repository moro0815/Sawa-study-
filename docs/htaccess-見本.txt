# ============================================================
#  【これは「見本」です。ふつうは使わなくて大丈夫です】
#
#  沙和ナビは、このファイルが無くても問題なく動きます。
#  以前は同梱していましたが、更新のたびに上書きすると
#  エックスサーバーの「アクセス制限」(パスワード)が
#  外れてしまうため、同梱をやめました。
#
#  ・パスワード      → サーバーパネルの「アクセス制限」でかけてください
#  ・HTTPSへの誘導   → アプリ側(JavaScript)で自動的に行います
#  ・検索よけ        → robots.txt と各HTMLの meta で効いています
#
#  それでも細かく設定したい方だけ、この中身を
#  public_html/.htaccess に「追記」してください(上書きは禁物です)。
#  追記したあとは、アクセス制限を一度OFF→ONし直してください。
# ============================================================

# ── 1. 常時HTTPS(必須) ────────────────────────────────
#  Service Worker(オフライン起動)、共有シート、クリップボードは
#  すべて HTTPS でないと動きません。
<IfModule mod_rewrite.c>
  RewriteEngine On
  RewriteCond %{HTTPS} off
  RewriteCond %{HTTP:X-Forwarded-Proto} !https
  RewriteRule ^(.*)$ https://%{HTTP_HOST}%{REQUEST_URI} [R=301,L]
</IfModule>

# ── 2. MIMEタイプ ────────────────────────────────────
<IfModule mod_mime.c>
  AddType text/javascript .js
  AddType image/png       .png
  AddCharset UTF-8        .html .css .js .json
</IfModule>
#  manifest.json だけを PWA 用の型にする。
#  (.json 全体に付けると将来ふつうのJSONを置いたとき壊れるため)
<Files "manifest.json">
  ForceType application/manifest+json
</Files>

# ── 3. キャッシュ ────────────────────────────────────
#  sw.js と manifest.json をキャッシュさせると更新が届かなくなります。
<IfModule mod_headers.c>
  <FilesMatch "^(sw\.js|manifest\.json)$">
    Header set Cache-Control "no-cache, no-store, must-revalidate"
    Header set Pragma "no-cache"
  </FilesMatch>
  <FilesMatch "\.(html)$">
    Header set Cache-Control "no-cache, must-revalidate"
  </FilesMatch>
  <FilesMatch "\.(css|js)$">
    Header set Cache-Control "public, max-age=3600"
  </FilesMatch>
  <FilesMatch "\.(png|ico|svg)$">
    Header set Cache-Control "public, max-age=604800"
  </FilesMatch>

  # ── 4. セキュリティヘッダ ──────────────────────────
  Header set X-Content-Type-Options "nosniff"
  Header set Referrer-Policy "no-referrer"
  Header set X-Frame-Options "DENY"
  # 検索エンジンに載せない(家族用のため)
  Header set X-Robots-Tag "noindex, nofollow, noarchive"

  # ── 4-2. ほかのサイトに埋め込ませない ──────────────
  #  中身の制限(script-src / connect-src など)は index.html と
  #  share.html の <meta> 側に書いてあります。ここに同じものを
  #  書くと、通信先を足したいときに2か所直すことになるので書きません。
  #  frame-ancestors だけは <meta> では効かないので、ここに置きます。
  Header set Content-Security-Policy "frame-ancestors 'none'"

  # ── 4-3. 使わない機能を止める ──────────────────────
  #  カメラ(宿題の撮影)とマイク(発音練習)は自分のページだけ許可。
  #  位置情報などは、うっかり使えないように完全に止めます。
  Header set Permissions-Policy "camera=(self), microphone=(self), geolocation=(), payment=(), usb=(), interest-cohort=()"
</IfModule>

# ── 5. ファイル一覧を見せない ─────────────────────────
Options -Indexes

# ── 6. 不要ファイルへのアクセスを塞ぐ ───────────────────
<FilesMatch "^(\.htaccess|\.git|.*\.md|package.*\.json)$">
  Require all denied
</FilesMatch>
<IfModule !mod_authz_core.c>
  <FilesMatch "^(\.htaccess|\.git|.*\.md)$">
    Order allow,deny
    Deny from all
  </FilesMatch>
</IfModule>

# ── 7. 圧縮 ─────────────────────────────────────────
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css text/javascript application/javascript application/json
</IfModule>

# ── 8. 文字化け防止 ──────────────────────────────────
<IfModule mod_headers.c>
  <FilesMatch "\.html$">
    Header set Content-Type "text/html; charset=UTF-8"
  </FilesMatch>
</IfModule>
